Tutorial — "Living Dream" (pentatonic phrase generator)

A Brief Introduction to this Synthesizer

The generator transforms the textual name of a dream into musical "fabric" on a pentatonic scale. The name provides a semantic clue (base, octave, color, mood), and also serves as a source of randomness through a deterministic seed—the same text always produces the same sound. The result is a set of short phrases that are stitched together with a crossfade, formed with an ADSR envelope, and can be saved as a WAV file.

What you'll hear and what parameters you can change

  • Dream Name — changes the base note and "character" (semantics: water/night/flight, etc.).
  • Mode — "Full Dream" (several phrases in a row) or "Musical Phrase" (one short phrase).
  • Crossfade — the duration of the overlap between phrases; At higher values, phrases flow smoothly.
  • Attack / Release — ADSR section, affects the pleasantness of the beginning and end of the sound.
  • Tempo — time multiplier, speeds up or slows down the pulse of phrases.
  • The Save WAV button exports the result to a .wav file.
Experiment idea: enter the same name, but change only the tempo — you will see how the same "melody" changes the sense of time. Another experiment: increase the crossfade — the result will resemble a long ambient landscape.

2) Deterministic Random Number Generator


// Convert the string to a numeric seed and create a PRNG
function stringToSeed(str){
let s = 0;
for (let c of str) s = (s * 31 + c.charCodeAt(0)) >>> 0;
return s;
}
function seededRandom(seed){
return () => {
seed ^= seed << 13;
seed ^= seed >>> 17;
seed ^= seed << 5;
return ((seed >>> 0) % 10000) / 10000;
};
}

3) Generating a single pentatonic phrase


// baseFreq — base note frequency (in Hz)
// duration — approximate phrase time (sec)
// rand — random number generator function
// attack/release — envelope parameters
function generatePhrase(baseFreq, duration, rand, attack, release, tempo){
const SAMPLE_RATE = 44100;
const len ​​= Math.floor(duration * SAMPLE_RATE);
const data = new Float32Array(len);
const pent = [0,2,4,7,9]; // pentatonic intervals (in semitones)
let pos = 0;
while (pos < len) { 
const note = pent[Math.floor(rand() * pent.length)]; 
const freq = baseFreq * Math.pow(2, note / 12) * tempo; 
const noteLen = Math.floor(SAMPLE_RATE * (0.25 + rand() * 0.25)); // length of a single note 
for (let i = 0; i < noteLen && pos < len; i++, pos++) { 
const t = i / noteLen; 
const env = (t < attack ? t / attack : t > 1 - release ? (1 - t) / release : 1); 
data[pos] = Math.sin(2 * Math.PI * freq * (pos / SAMPLE_RATE)) * 0.35 * env; 
} 
} 
return data;
}

4) Merging phrases with crossfade


// phrases — Float32Array
// crossfade — duration in seconds
function mergePhrases(phrases, crossfade){
const SAMPLE_RATE = 44100;
const fadeSamples = Math.floor(crossfade * SAMPLE_RATE);
let total = phrases.reduce((a,b)=>a+b.length,0) - (phrases.length - 1) * fadeSamples;
const out = new Float32Array(total);
let pos = 0;
for (let i = 0; i < phrases.length; i++) {
const p = phrases[i];
if (i > 0) { 
for (let j = 0; j < fadeSamples; j++) { 
const t = j / fadeSamples; 
out[pos - fadeSamples + j] = out[pos - fadeSamples + j] * (1 - t) + p[j] * t; 
} 
out.set(p.subarray(fadeSamples), pos); 
pos += p.length - fadeSamples; 
} else { 
out.set(p, pos); 
pos += p.length; 
} 
} 
return out;
}

5) Export to WAV (short implementation)


// Returns an ArrayBuffer with WAV data (16-bit PCM)
function encodeWav(samples){ 
const SAMPLE_RATE = 44100; 
const buffer = new ArrayBuffer(44 + samples.length * 2); 
const view = new DataView(buffer); 
function writeString(o, s){ for (let i = 0; i < s.length; i++) view.setUint8(o + i, s.charCodeAt(i)); } 
writeString(0, "RIFF"); 
view.setUint32(4, 36 + samples.length * 2, true); 
writeString(8, "WAVE"); 
writeString(12, "fmt "); 
view.setUint32(16, 16, true);view.setUint16(20, 1, true); 
view.setUint16(22, 1, true); 
view.setUint32(24, SAMPLE_RATE, true); 
view.setUint32(28, SAMPLE_RATE * 2, true); 
view.setUint16(32, 2, true); 
view.setUint16(34, 16, true); 
writeString(36, "data"); 
view.setUint32(40, samples.length * 2, true); 
let offset = 44; 
for (let i = 0; i < samples.length; i++, offset += 2) { 
let s = Math.max(-1, Math.min(1, samples[i])); 
view.setInt16(offset, s < 0 ? s * 0x8000 : s * 0x7FFF, true);
}
return buffer;
}

Recommendations and Improvements

  • If you want a more pronounced bass, lower baseFreq by an octave for some phrases.
  • For complex themes, replace the sine wave with a harmonic mixture (additive synthesis) – this will make the phrase richer.
  • Visualization (line state) can be simplified. If the canvas is loading the device, reduce the length of the buffer passed to the drawing program.
  • WAV export – works locally in the browser; for longer durations, a progress bar may be required.
