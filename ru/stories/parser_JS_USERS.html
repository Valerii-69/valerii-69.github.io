Лабораторная работа — проверка JS-кода

Инструкция для читателя:

  • Используйте только переданный canvas, не обращайтесь к document.getElementById.
  • Ваш код должен быть в формате функции, принимающей canvas:
    • (function(canvas){ /* ваш код */ })(canvas);
  • Все переменные должны быть локальными, анимации — через requestAnimationFrame или setInterval.

Пример рабочего кода:

// 0. Бегущий прямоугольник 
(function(canvas){
  const ctx = canvas.getContext('2d');
  ctx.clearRect(0, 0, canvas.width, canvas.height);
  let x = 0;
  function animate(){
    ctx.clearRect(0,0,canvas.width,canvas.height);
    ctx.fillStyle='red';
    ctx.fillRect(x,90,50,30);
    x = (x + 2) % canvas.width;
    requestAnimationFrame(animate);
  }
  animate();
})(canvas);

// 1. Фрактальное дерево
(function(canvas){
const treeCtx = canvas.getContext('2d');
  treeCtx.clearRect(0, 0, canvas.width, canvas.height);
function drawTree(x, y, length, angle, depth) {
  if(depth===0) return;
  const x1 = x + length * Math.cos(angle);
  const y1 = y - length * Math.sin(angle);
  treeCtx.beginPath();
  treeCtx.moveTo(x, y);
  treeCtx.lineTo(x1, y1);
  treeCtx.stroke();
  drawTree(x1, y1, length*0.7, angle - Math.PI/6, depth-1);
  drawTree(x1, y1, length*0.7, angle + Math.PI/6, depth-1);
}
drawTree(280, 190, 60, Math.PI/2, 8);
})(canvas);

// 2. Снежный фрактал (Кривая Коха)
(function(canvas){
const kochCtx = canvas.getContext('2d');
  kochCtx.clearRect(0, 0, canvas.width, canvas.height);
function kochLine(x1,y1,x2,y2,depth){
  if(depth===0){kochCtx.beginPath();kochCtx.moveTo(x1,y1);kochCtx.lineTo(x2,y2);kochCtx.stroke();return;}
  const dx = (x2-x1)/3;
  const dy = (y2-y1)/3;
  const xA = x1 + dx, yA = y1 + dy;
  const xB = 0.5*(x1+x2) - Math.sqrt(3)*(y2-y1)/6, yB = 0.5*(y1+y2) + Math.sqrt(3)*(x2-x1)/6;
  const xC = x1 + 2*dx, yC = y1 + 2*dy;
  kochLine(x1,y1,xA,yA,depth-1);
  kochLine(xA,yA,xB,yB,depth-1);
  kochLine(xB,yB,xC,yC,depth-1);
  kochLine(xC,yC,x2,y2,depth-1);
}
kochLine(120,50,400,50,3);
})(canvas);
