Учебник — «Живой сон» (генератор пентатонических фраз)

Коротко — о чём этот синтезатор

Генератор превращает текстовое имя сна в музыкальную «ткань» на пентатонической гамме. Имя даёт семантическую подсказку (база, октава, цвет, настроение), а также служит источником случайности через детерминированный seed — один и тот же текст всегда даёт одно и то же звучание. Итог — набор коротких фраз, которые склеиваются с кроссфейдом, оформляются ADSR-огибающей и могут быть сохранены в WAV.

Что вы услышите и какие параметры можно менять

  • Имя сна — меняет базовую ноту и «характер» (семантика: вода/ночь/полет и т.п.).
  • Mode — «Полный сон» (несколько фраз подряд) или «Музыкальная фраза» (одна короткая фраза).
  • Crossfade — длительность перекрытия между фразами; при больших значениях фразы плавно перетекают.
  • Attack / Release — ADSR-часть, влияет на приятность начала и концов звучания.
  • Tempo — множитель времени, ускоряет или замедляет пульс фраз.
  • Кнопка Сохранить WAV экспортирует итог в файл .wav.
Идея для эксперимента: введите одно и то же имя, но меняйте только tempo — вы увидите, как та же «мелодия» меняет ощущение времени. Другой эксперимент — увеличить crossfade — результат станет похож на длинный амбиентный пейзаж.

2) Детерминированный генератор случайных чисел


// Преобразуем строку в числовой seed и создаём PRNG
function stringToSeed(str){
  let s = 0;
  for (let c of str) s = (s * 31 + c.charCodeAt(0)) >>> 0;
  return s;
}
function seededRandom(seed){
  return () => {
    seed ^= seed << 13;
    seed ^= seed >>> 17;
    seed ^= seed << 5;
    return ((seed >>> 0) % 10000) / 10000;
  };
}

3) Генерация одной пентатонической фразы


// baseFreq — частота базовой ноты (в Гц)
// duration — примерное время фразы (сек)
// rand — функция генератора случайных чисел
// attack/release — параметры огибающей
function generatePhrase(baseFreq, duration, rand, attack, release, tempo){
  const SAMPLE_RATE = 44100;
  const len = Math.floor(duration * SAMPLE_RATE);
  const data = new Float32Array(len);
  const pent = [0,2,4,7,9]; // интервалы пентатоники (в полутонов)
  let pos = 0;
  while (pos < len) {
    const note = pent[Math.floor(rand() * pent.length)];
    const freq = baseFreq * Math.pow(2, note / 12) * tempo;
    const noteLen = Math.floor(SAMPLE_RATE * (0.25 + rand() * 0.25)); // длина отдельной ноты
    for (let i = 0; i < noteLen && pos < len; i++, pos++) {
      const t = i / noteLen;
      const env = (t < attack ? t / attack : t > 1 - release ? (1 - t) / release : 1);
      data[pos] = Math.sin(2 * Math.PI * freq * (pos / SAMPLE_RATE)) * 0.35 * env;
    }
  }
  return data;
}

4) Склейка фраз с кроссфейдом


// phrases — массив Float32Array
// crossfade — длительность в секундах
function mergePhrases(phrases, crossfade){
  const SAMPLE_RATE = 44100;
  const fadeSamples = Math.floor(crossfade * SAMPLE_RATE);
  let total = phrases.reduce((a,b)=>a+b.length,0) - (phrases.length - 1) * fadeSamples;
  const out = new Float32Array(total);
  let pos = 0;
  for (let i = 0; i < phrases.length; i++) {
    const p = phrases[i];
    if (i > 0) {
      for (let j = 0; j < fadeSamples; j++) {
        const t = j / fadeSamples;
        out[pos - fadeSamples + j] = out[pos - fadeSamples + j] * (1 - t) + p[j] * t;
      }
      out.set(p.subarray(fadeSamples), pos);
      pos += p.length - fadeSamples;
    } else {
      out.set(p, pos);
      pos += p.length;
    }
  }
  return out;
}

5) Экспорт в WAV (короткая реализация)


// Возвращает ArrayBuffer с WAV-данными (16-bit PCM)
function encodeWav(samples){
  const SAMPLE_RATE = 44100;
  const buffer = new ArrayBuffer(44 + samples.length * 2);
  const view = new DataView(buffer);
  function writeString(o, s){ for (let i = 0; i < s.length; i++) view.setUint8(o + i, s.charCodeAt(i)); }
  writeString(0, "RIFF");
  view.setUint32(4, 36 + samples.length * 2, true);
  writeString(8, "WAVE");
  writeString(12, "fmt ");
  view.setUint32(16, 16, true);
  view.setUint16(20, 1, true);
  view.setUint16(22, 1, true);
  view.setUint32(24, SAMPLE_RATE, true);
  view.setUint32(28, SAMPLE_RATE * 2, true);
  view.setUint16(32, 2, true);
  view.setUint16(34, 16, true);
  writeString(36, "data");
  view.setUint32(40, samples.length * 2, true);
  let offset = 44;
  for (let i = 0; i < samples.length; i++, offset += 2) {
    let s = Math.max(-1, Math.min(1, samples[i]));
    view.setInt16(offset, s < 0 ? s * 0x8000 : s * 0x7FFF, true);
  }
  return buffer;
}

Рекомендации и улучшения

  • Если хотите более выраженный бас — понизьте baseFreq на октаву для части фраз.
  • Для сложных тем замените синус на смесь гармоник (additive synthesis) — так фраза станет богаче.
  • Визуализация (line-стейт) можно упростить, если canvas нагружает устройство — уменьшите длину буфера, передаваемого в рисовалку.
  • Экспорт WAV — работает локально в браузере; при больших длительностях может потребоваться прогресс-бар.
